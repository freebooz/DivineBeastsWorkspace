cmake_minimum_required(VERSION 3.20)
project(GamePlatformAbilityOwnershipTests LANGUAGES CXX)
enable_testing()
add_executable(AbilityOwnershipTests ../Source/GamePlatformAbilitySystem/Private/Tests/AbilityOwnershipTests.cpp)
target_compile_features(AbilityOwnershipTests PRIVATE cxx_std_20)
target_compile_definitions(AbilityOwnershipTests PRIVATE PLATFORM_ABILITY_NATIVE_TESTS=1)
if(MSVC)
  target_compile_options(AbilityOwnershipTests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(AbilityOwnershipTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME AbilityOwnershipTests COMMAND AbilityOwnershipTests)
