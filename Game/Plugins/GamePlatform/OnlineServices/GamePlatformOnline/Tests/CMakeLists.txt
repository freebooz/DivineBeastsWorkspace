cmake_minimum_required(VERSION 3.20)
# 仅编译历史纯逻辑回归模型与测试，不进入 UE 正式运行时；用于保留策略迁移基线，不证明 UHT／UE 链接通过。
project(GamePlatformOnlineLogicTests LANGUAGES CXX)
enable_testing()
set(PRIVATE_DIR "${CMAKE_CURRENT_LIST_DIR}/../Source/GamePlatformOnline/Private")
add_executable(OnlineLogicTests "${PRIVATE_DIR}/Tests/OnlineSessionTests.cpp" "${PRIVATE_DIR}/Requests/OnlineSession.cpp")
target_include_directories(OnlineLogicTests PRIVATE "${PRIVATE_DIR}")
target_compile_features(OnlineLogicTests PRIVATE cxx_std_17)
target_compile_definitions(OnlineLogicTests PRIVATE GAMEPLATFORM_ONLINE_NATIVE_TEST=1)
if(MSVC)
  target_compile_options(OnlineLogicTests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(OnlineLogicTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME OnlineLogicTests COMMAND OnlineLogicTests)
