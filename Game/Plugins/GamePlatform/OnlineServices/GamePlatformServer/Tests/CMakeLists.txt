# 平台服务器便携回归入口：编译生产接收预算；UE所有权测试位于Private/Tests，由正式目标编译。
cmake_minimum_required(VERSION 3.20)
project(GamePlatformAdmissionNativeTests LANGUAGES CXX)
enable_testing()
add_executable(AdmissionResponseBudgetTests ../Source/GamePlatformServer/Private/Tests/AdmissionResponseBudgetTests.cpp)
target_compile_definitions(AdmissionResponseBudgetTests PRIVATE GAMEPLATFORM_ADMISSION_NATIVE_TEST=1)
target_include_directories(AdmissionResponseBudgetTests PRIVATE ../Source/GamePlatformServer/Private)
target_compile_features(AdmissionResponseBudgetTests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(AdmissionResponseBudgetTests PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(AdmissionResponseBudgetTests PRIVATE -UNDEBUG)
endif()
add_test(NAME AdmissionResponseBudget COMMAND AdmissionResponseBudgetTests)
